cmake_minimum_required(VERSION 3.25)
project(owcs-ob-analyzer VERSION 0.5.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This release targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN QT_NO_KEYWORDS)
add_compile_options(/utf-8 /EHsc /W3)
set(OBS_INSTALL "" CACHE PATH "OBS Studio installation directory")
set(DEPS_DIR "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Dependency directory")
set(OBS_SOURCE "${DEPS_DIR}/obs/obs-studio-32.2.2")
set(ORT_ROOT "${DEPS_DIR}/ort/onnxruntime-win-x64-1.20.1")
set(OpenCV_DIR "${DEPS_DIR}/opencv/opencv/build/x64/vc16/lib")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS_DIR}/qt6")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Network)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs)
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/obsconfig.h")
add_library(owcs-core STATIC native/core/recognizer.cpp native/core/life.cpp native/core/names.cpp native/core/state.cpp native/core/public_state.cpp)
target_include_directories(owcs-core PUBLIC native "${ORT_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(owcs-core PUBLIC Qt6::Core "${DEPS_DIR}/obs-sdk/owcs-opencv4100.lib" "${DEPS_DIR}/obs-sdk/owcs-onnxruntime.lib")
add_library(owcs-runtime STATIC native/runtime/engine.cpp native/runtime/input.cpp native/runtime/registry.cpp native/runtime/server.cpp)
target_link_libraries(owcs-runtime PUBLIC owcs-core Qt6::Network user32 advapi32)
add_library(owcs-ob-analyzer MODULE native/obs/plugin.cpp native/obs/filter.cpp native/obs/dock.cpp)
target_include_directories(owcs-ob-analyzer PRIVATE "${OBS_SOURCE}/libobs" "${OBS_SOURCE}/frontend/api" "${CMAKE_BINARY_DIR}")
target_link_libraries(owcs-ob-analyzer PRIVATE owcs-runtime Qt6::Widgets "${DEPS_DIR}/obs-sdk/obs.lib" "${DEPS_DIR}/obs-sdk/obs-frontend-api.lib")
set_target_properties(owcs-ob-analyzer PROPERTIES PREFIX "")
add_executable(owcs-recognizer-test native/tests/main.cpp native/tests/life_regression.cpp native/tests/state_regression.cpp)
target_link_libraries(owcs-recognizer-test PRIVATE owcs-runtime)
add_custom_command(TARGET owcs-recognizer-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:owcs-recognizer-test>/owcs-onnxruntime.dll")
add_custom_command(TARGET owcs-recognizer-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DEPS_DIR}/opencv/opencv/build/x64/vc16/bin/opencv_world4100.dll" "$<TARGET_FILE_DIR:owcs-recognizer-test>/owcs-opencv4100.dll")
enable_testing()
add_test(NAME native-regression COMMAND owcs-recognizer-test "${CMAKE_SOURCE_DIR}/data/models" "${CMAKE_SOURCE_DIR}/tests/data/scoreboard")
set_tests_properties(native-regression PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OBS_INSTALL}/bin/64bit;PATH=path_list_prepend:${DEPS_DIR}/qt6/bin;PATH=path_list_prepend:${DEPS_DIR}/opencv/opencv/build/x64/vc16/bin;PATH=path_list_prepend:${ORT_ROOT}/lib")
install(TARGETS owcs-ob-analyzer LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
install(FILES "${DEPS_DIR}/opencv/opencv/build/x64/vc16/bin/opencv_world4100.dll" DESTINATION obs-plugins/64bit RENAME owcs-opencv4100.dll)
install(FILES "${ORT_ROOT}/lib/onnxruntime.dll" DESTINATION obs-plugins/64bit RENAME owcs-onnxruntime.dll)
install(DIRECTORY data/ DESTINATION data/obs-plugins/owcs-ob-analyzer)
